const XP_ITEM_EFFECTS = {
  "経験の書": {
    multiplier: 1.5,
    durationMinutes: 60,
  },

  "上級経験の書": {
    multiplier: 2,
    durationMinutes: 30,
  },

  "極経験の書": {
    multiplier: 3,
    durationMinutes: 15,
  },

  "長時間経験の書": {
    multiplier: 1.5,
    durationMinutes: 90,
  },

  "超長時間経験の書": {
    multiplier: 1.5,
    durationMinutes: 120,
  },

  "連続の心得": {
    multiplier: 1.2,
    durationMinutes: null,
  },
};

function calculateStudyXp(minutes, itemName) {
  const baseMinutes = Math.max(0, Number(minutes || 0));

  if (!itemName || !XP_ITEM_EFFECTS[itemName]) {
    return baseMinutes;
  }

  const effect = XP_ITEM_EFFECTS[itemName];

  // 連続の心得：次のタイマー全体
  if (effect.durationMinutes == null) {
    return Math.floor(
      baseMinutes * effect.multiplier
    );
  }

  // その他：効果時間内だけ倍率
  const boostedMinutes =
    Math.min(baseMinutes, effect.durationMinutes);

  const normalMinutes =
    Math.max(0, baseMinutes - boostedMinutes);

  return (
    Math.floor(boostedMinutes * effect.multiplier) +
    normalMinutes
  );
}
